# FreqUI 统计分析模块集成方案

> 版本：v1.0 | 日期：2026-05-12 | 状态：待确认

---

## 一、项目背景

### 1.1 现状

当前系统由两个独立服务组成：

| 服务 | 技术栈 | 端口 | 作用 |
|------|--------|------|------|
| freqtrade | Python + FastAPI | 8080 | 交易机器人 + REST API + 内置 FreqUI |
| dashboard | Python + FastAPI + Bootstrap 5 | 8788 | 统计分析（交易查询、实盘报告、回测对比、选币扫描） |

**痛点：**
- 需要同时启动两个服务，运维复杂
- 两套 UI 风格不统一（FreqUI 用 Vue + PrimeVue，Dashboard 用 Bootstrap 5）
- Dashboard 需要手动输入初始资金、手动切换数据库文件
- 无法利用 freqtrade 已有的配置信息（交易所、余额、运行模式等）

### 1.2 目标

将 Dashboard 的全部功能集成到 freqtrade + FreqUI 体系中：

- **一个服务**：所有 API 由 freqtrade 统一提供（端口 8080）
- **一套 UI**：FreqUI 导航栏新增「📊 统计分析」菜单，风格统一
- **自动化**：自动读取 bot 配置，减少手动输入
- **向后兼容**：保留高级功能（切换数据库、手动输入余额）作为可选项

FreqUI 统计分析 API 路由
前缀: /api/v1/dashboard

from fastapi import APIRouter, BackgroundTasks, HTTPException, Query, Request, UploadFile, File
from typing import Optional

router = APIRouter(prefix="/dashboard", tags=["Dashboard"])


# ========== 交易历史 ==========

@router.get("/trades")
async def dashboard_trades(
    request: Request,
    pair: Optional[str] = Query(None),
    exit_reason: Optional[str] = Query(None),
    enter_tag: Optional[str] = Query(None),
    is_short: Optional[bool] = Query(None),
    date_from: Optional[str] = Query(None),
    date_to: Optional[str] = Query(None),
    min_profit: Optional[float] = Query(None),
    max_profit: Optional[float] = Query(None),
    limit: int = Query(500),
    offset: int = Query(0),
):
    """
    查询历史交易记录（已平仓）
    支持按交易对、方向、标签、时间、收益范围筛选
    """
    svc = request.app.state.dashboard_trade_service
    filters = {
        "pair": pair, "exit_reason": exit_reason, "enter_tag": enter_tag,
        "is_short": is_short, "date_from": date_from, "date_to": date_to,
        "min_profit": min_profit, "max_profit": max_profit,
        "limit": limit, "offset": offset,

    return {"trades": trades, "stats": stats}

async def bot_info(request: Request):
    """
    获取 bot 自动检测信息（新增接口）
    让前端知道自动获取了哪些值
    """
    config = request.app.state.ft_config
    rpc = getattr(request.app.state, "_rpc", None)

    info = {
        "exchange": config.get("exchange", "unknown"),
        "stake_currency": config.get("stake_currency", "USDT"),
        "trading_mode": config.get("trading_mode", "spot"),
        "run_mode": config.get("runmode", "unknown"),
        "db_path": str(request.app.state.dashboard_trade_service.db_path),
        "starting_balance": _resolve_starting_balance(request),
        "dry_run": config.get("dry_run", True),
    }

    # 如果有 RPC，尝试获取实时余额
    if rpc:
        try:
            balance = rpc._rpc_balance()
            info["current_balance"] = balance.get("total", 0)
        except Exception:
            pass

    return info


# ========== 内部工具函数 ==========

def _resolve_starting_balance(request: Request) -> float:
    """
    自动解析初始资金，优先级：
    1. bot config 中的 dry_run_wallet
    2. 实盘模式下的实际总资产
    3. 默认值 1000
    """
    config = request.app.state.ft_config

    # 模拟盘：读 dry_run_wallet
    if config.get("dry_run", True):
        return config.get("dry_run_wallet", 1000.0)

    # 实盘：尝试读余额
    rpc = getattr(request.app.state, "_rpc", None)
    if rpc:
        try:
            balance = rpc._rpc_balance()
            return balance.get("total", 1000.0)
        except Exception:
            pass

    return 1000.0

### 4.3 `stores/dashboardStore.ts` — Store 定义

```typescript
import { defineStore } from "pinia";
import axios from "axios";
import type {
  DashboardTrade,
  TradeStats,
  TradeFilters,
  TradeQueryParams,
  ScanResult,
  ReportStatus,
  CompareResult,
  BotInfo,
} from "@/types/dashboard";

const API_BASE = "/api/v1/dashboard";

export const useDashboardStore = defineStore("dashboard", {
  state: () => ({
    // 交易历史
    trades: [] as DashboardTrade[],
    tradeStats: { total: 0, total_pnl: 0, avg_profit_pct: 0, winrate: 0 } as TradeStats,
    filters: { exit_reasons: [], enter_tags: [], pairs: [] } as TradeFilters,
    tradesLoading: false,

    // 选币扫描
    scanResult: null as ScanResult | null,
    scanLoading: false,

    // 实盘报告
    reportStatus: "idle" as string,
    reportLog: "",
    reportData: null as any,
    reportUpdatedAt: null as string | null,

    // Bot 信息（自动检测）
    botInfo: null as BotInfo | null,
  }),

  getters: {
    scanData: (state) => state.scanResult?.data || [],
    scanUpdatedAt: (state) => state.scanResult?.updated_at || "-",
    scanCount: (state) => state.scanResult?.count || 0,
    reportReady: (state) => state.reportStatus === "ok" && state.reportData !== null,
  },

  actions: {
    // ========== 交易历史 ==========

    async fetchTrades(params: TradeQueryParams = {}) {
      this.tradesLoading = true;
      try {
        const { data } = await axios.get(`${API_BASE}/trades`, { params });
        this.trades = data.trades;
        this.tradeStats = data.stats;
      } catch (err) {
        console.error("Failed to fetch trades:", err);
      } finally {
        this.tradesLoading = false;
      }
    },

    async fetchFilters() {
      try {
        const { data } = await axios.get(`${API_BASE}/trades/filters`);
        this.filters = data;
      } catch (err) {
        console.error("Failed to fetch filters:", err);
      }
    },

    // ========== 选币扫描 ==========

    async fetchScan() {
      try {
        const { data } = await axios.get(`${API_BASE}/market-scan`);
        this.scanResult = data;
      } catch (err) {
        console.error("Failed to fetch scan:", err);
      }
    },

    async refreshScan() {
      this.scanLoading = true;
      try {
        await axios.post(`${API_BASE}/market-scan/refresh`);
        // 轮询状态直到完成
        const poll = setInterval(async () => {
          await this.fetchScan();
          if (this.scanResult?.status !== "running") {
            clearInterval(poll);
            this.scanLoading = false;
          }
        }, 3000);
      } catch (err) {
        console.error("Failed to refresh scan:", err);
        this.scanLoading = false;
      }
    },

    // ========== 实盘报告 ==========

    async fetchBotInfo() {
      try {
        const { data } = await axios.get(`${API_BASE}/config/bot-info`);
        this.botInfo = data;
      } catch (err) {
        console.error("Failed to fetch bot info:", err);
      }
    },

    async runReport(balance?: number) {
      try {
        const params: any = {};
        if (balance !== undefined) params.starting_balance = balance;
        await axios.post(`${API_BASE}/live-report/run`, null, { params });
        this.reportStatus = "running";
        // 轮询
        const poll = setInterval(async () => {
          await this.fetchReportStatus();
          if (this.reportStatus !== "running") {
            clearInterval(poll);
            if (this.reportStatus === "ok") {
              await this.fetchReportData();
            }
          }
        }, 3000);
      } catch (err) {
        console.error("Failed to run report:", err);
      }
    },

    async fetchReportStatus() {
      try {
        const { data } = await axios.get(`${API_BASE}/live-report/status`);
        this.reportStatus = data.status;
        this.reportLog = data.log;
        this.reportUpdatedAt = data.updated_at;
      } catch (err) {
        console.error("Failed to fetch report status:", err);
      }
    },

    async fetchReportData() {
      try {
        const { data } = await axios.get(`${API_BASE}/live-report/data`);
        this.reportData = data.data;
        this.reportStatus = data.status;
      } catch (err) {
        console.error("Failed to fetch report data:", err);
      }
    },

    // ========== 回测对比 ==========

    async compareBacktest(liveFile: File, btFile: File): Promise<CompareResult | null> {
      try {
        const formData = new FormData();
        formData.append("live_file", liveFile);
        formData.append("bt_file", btFile);
        const { data } = await axios.post(`${API_BASE}/backtest/compare`, formData, {
          headers: { "Content-Type": "multipart/form-data" },
        });
        return data;
      } catch (err) {
        console.error("Failed to compare backtest:", err);
        return null;
      }
    },

    // ========== 数据库切换 ==========

    async fetchDatabases() {
      try {
        const { data } = await axios.get(`${API_BASE}/config/databases`);
        return data;
      } catch (err) {
        console.error("Failed to fetch databases:", err);
        return { databases: [], environment: "unknown", active_path: "" };
      }
    },

    async switchDatabase(dbName: string) {
      try {
        await axios.post(`${API_BASE}/config/switch-db`, null, {
          params: { db_name: dbName },
        });
        // 切换后刷新数据
        await this.fetchTrades();
        await this.fetchFilters();
      } catch (err) {
        console.error("Failed to switch database:", err);
      }
    },
  },
});
```

| 认证 | 无 | 复用 freqtrade 的 JWT/HTTP Basic |
| 数据库切换 | 保留 | 保留（高级功能，查看其他策略库） |

## 七、风险与注意事项

| 风险 | 说明 | 应对措施 |
|------|------|---------|
| `live_report.py` 脚本依赖 | 脚本可能依赖特定路径或 Python 包 | 迁移时检查所有 import 路径 |
| MarketScanner 的网络请求 | 调用 Binance API，可能超时或被限流 | 已有 REST 降级方案，超时设置 30s |
| 大数据库查询性能 | SQLite 在大数据量下可能较慢 | 保留 LIMIT/OFFSET 分页 |
| 前端构建版本兼容 | FreqUI 和 freqtrade 版本需匹配 | 部署前确认 API 版本 |
| 回测对比文件大小 | 上传大文件可能超时 | 前端限制文件大小，后端设置超时 |
